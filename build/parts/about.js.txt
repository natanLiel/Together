  // the picked tags, in one order for everyone, each with its icon
  aboutTags(about, he) {
    const a = about || {}, tr = s => he ? (Component.ABOUT_HE[s] || s) : s, out = [];
    Component.ABOUT.forEach(c => {
      const v = a[c.k];
      if (c.multi) { if (v && v.length) out.push({ label: v.map(tr).join(', '), icon: c.icon }); return; }
      if (!v) return;
      out.push({ label: tr(v), icon: c.k === 'rhythm' && v === 'Night owl' ? Component.ABOUT_MOON : c.icon });
    });
    if (a.own && a.own.trim()) out.push({ label: a.own.trim(), icon: Component.ABOUT_OWN_ICON });
    return out;
  }

  setAbout(k, v) {
    const me = this.state.me, about = Object.assign({}, me.about || {});
    const c = Component.ABOUT.find(x => x.k === k);
    if (c && c.multi) { const cur = (about[k] || []).slice(), i = cur.indexOf(v); if (i >= 0) cur.splice(i, 1); else cur.push(v); about[k] = cur; }
    else if (c) about[k] = about[k] === v ? '' : v;
    else about[k] = v;
    this.setState({ me: Object.assign({}, me, { about }) });
  }
